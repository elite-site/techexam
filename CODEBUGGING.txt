 Buggy Code

#include <stdlib.h>
#include <string.h>

struct Student {
    char name[30];
    int marks[3];
    float average;
};

int main() {
    int n, i, j;
    float total;
    struct Student *s;

    printf("Enter number of students: ");
    scanf("%d", &n);

    s = (struct Student *)malloc(n * sizeof(struct Student));

    if (s == NULL) {
        printf("Memory allocation failed\n");
        return 1;
    }

    for (i = 0; i < n; i++) {
        printf("\nEnter name: ");
        scanf("%29s", s[i].name);

        total = 0;

        printf("Enter 3 marks: ");
        for (j = 0; j < 3; j++) {
            scanf("%d", &s[i].marks[j]);
            total += s[i].marks[j];
        }

        s[i].average = total / 3;
    }

    for (i = 0; i < n; i++) {
        if (s[i].average > 100) {              // ❌ ERROR 1
            s[i].average = 100;
        }
    }

    int highest = 0;

    for (i = 1; i < n; i++) {
        if (s[i].average > s[highest].average) {
            highest = i;
        }
    }

    printf("\nTop Student: %s\n", s[highest].name);
    printf("Average: %.2f\n", s[highest].average);

    printf("Top student: %s\n", s[highest].name);
